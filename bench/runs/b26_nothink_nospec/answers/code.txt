def merge_intervals(intervals):
    if not intervals:
        return []

    # 1. Сортируем интервалы по начальной точке
    intervals.sort(key=lambda x: x[0])

    merged = [intervals[0]]

    for current in intervals[1:]:
        last_merged = merged[-1]

        # 2. Если текущий интервал пересекается с последним добавленным — объединяем
        if current[0] <= last_merged[1]:
            last_merged[1] = max(last_merged[1], current[1])
        else:
            # 3. Если не пересекается — добавляем как новый интервал
            merged.append(current)

    return merged

# Пример использования:
example = [[1, 3], [2, 6], [8, 10], [15, 18]]
print(merge_intervals(example))  # Вывод: [[1, 6], [8, 10], [15, 18]]
