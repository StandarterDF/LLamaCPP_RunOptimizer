import csv
import json
from typing import List, Dict, Any

def load_data(path: str) -> List[Dict[str, Any]]:
    """Loads records from a CSV file."""
    records = []
    with open(path, newline='') as f:
        reader = csv.DictReader(f)
        for row in reader:
            records.append(row)
    return records

def clean_data(records: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """Cleans records by stripping whitespace from names."""
    out = []
    for item in records:
        if item.get('name') is None:
            continue
        item['name'] = item['name'].strip()
        out.append(item)
    return out

def aggregate(records: List[Dict[str, Any]]) -> Dict[str, int]:
    """Counts records by category."""
    result = {}
    for item in records:
        key = item.get('category', 'other')
        result[key] = result.get(key, 0) + 1
    return result
