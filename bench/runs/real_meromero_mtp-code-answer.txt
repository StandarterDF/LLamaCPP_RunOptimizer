Here is the Python function to merge overlapping intervals:

```python
def merge_intervals(intervals):
    if not intervals:
        return []

    # Sort intervals by their start time
    intervals.sort(key=lambda x: x[0])
    
    merged = [intervals[0]]
    
    for current in intervals[1:]:
        last_merged = merged[-1]
        
        # If the current interval overlaps with the last merged interval
        if current[0] <= last_merged[1]:
            # Update the end of the last merged interval to the max of both
            last_merged[1] = max(last_merged[1], current[1])
        else:
            # No overlap, add the current interval to the list
            merged.append(current)
            
    return merged

# Usage Example

print(f"Input: {intervals_list}")
print(f"Merged: {merge_intervals(intervals_list)}")
# Expected Output: [[1, 6], [8, 10], [15, 18]]
```
